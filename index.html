function exportarCatalogoAJSON() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Hoja 1"); // O el nombre exacto de tu pestaña var data = sheet.getDataRange().getValues(); var headers = data[0]; // Buscamos las columnas automáticamente por sus nombres var colDetalle = headers.indexOf("Detalle"); var colPrecio = headers.indexOf("Monto Venta 1"); var colCodigo = headers.indexOf("Codigo"); var productos = []; for (var i = 1; i < data.length; i++) { var row = data[i]; if (row[colDetalle]) { // Si tiene nombre/detalle productos.push({ codigo: row[colCodigo] || "", nombre: row[colDetalle], precio: row[colPrecio] || 0 }); } } var jsonOutput = JSON.stringify(productos, null, 2); Logger.log(jsonOutput); // Aquí se genera todo tu inventario en formato digital listo para GitHub // Opcional: Esto lo podemos conectar directo a GitHub con un token o descargarlo en un archivo JSON return jsonOutput; }
