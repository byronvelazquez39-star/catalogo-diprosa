function actualizarVistaMetas() { let tiquetesGuardados = JSON.parse(localStorage.getItem(ticketsKey)) || []; let totalVentasReal = 0, totalTq = 0; tiquetesGuardados.forEach(t => { if (t.total) { totalVentasReal += parseFloat(t.total); totalTq += 1; } else if (t.esAcumuladoBase && t.total) { totalVentasReal += parseFloat(t.total); totalTq += (t.tiquetesCount || 0); } }); document.getElementById('meta-ventas-real').innerText = totalVentasReal.toFixed(2); document.getElementById('meta-tiquetes-real').innerText = totalTq + " / 220 Tq Meta"; let contenedorLista = document.getElementById('lista-tiquetes'); if (!contenedorLista) contenedorLista = document.getElementById('lista-tiquetes-eleazar'); if (!contenedorLista) return; if (tiquetesGuardados.length === 0) { contenedorLista.innerHTML = '

No hay tiquetes registrados aún.

'; return; } contenedorLista.innerHTML = tiquetesGuardados.map((t, idx) => { if (t.esAcumuladoBase) { return `
📊 BASE DE INICIO (${t.tiquetesCount || 0} Tq)
Ventas Base: Q${t.total ? t.total.toFixed(2) : '0.00'}
`; } else { return `
Tiquete #${idx+1} - 👤 ${t.cliente || 'Cliente'}

`; } }).join(''); }
