/* CÁLCULO EN TIEMPO REAL 16+ (DÉFICIT, DESCUENTOS Y PAGOS) */ function abrirModalComisiones() { let ventasQ2 = 7190.04; // <- Sin comas (7190.04) let totalVentasMes = 39706.03; // <- Sin comas (39706.03) let sueldoBase = 3600; let metaMes = 81000; // Meta esperada a la fecha (8 días transcurridos del 16 al 23) let metaQ2Esperada = 8 * 2700; let deficitQ2 = metaQ2Esperada - ventasQ2; if (deficitQ2 < 0) deficitQ2 = 0; // Descuento proporcional sobre la base de la 2da quincena (Q1,800) let descuento = 0; let diaHoy = new Date().getDate(); if (diaHoy >= 16 && deficitQ2 > 0) { descuento = (deficitQ2 / metaQ2Esperada) * 1800; if (descuento > 1800) descuento = 1800; } // Comisión (solo si pasa de Q81,000) let comision = totalVentasMes > metaMes ? (totalVentasMes - metaMes) * 0.025 : 0; let bono = totalVentasMes >= metaMes ? 500 : 0; let totalNeto = (sueldoBase - descuento) + bono + comision; document.getElementById('q2-ventas-real').innerText = "Q " + ventasQ2.toFixed(2); document.getElementById('q2-deficit').innerText = "Q " + deficitQ2.toFixed(2); document.getElementById('q2-descuento').innerText = "Q " + descuento.toFixed(2); document.getElementById('q2-bono').innerText = bono > 0 ? "Q 500.00" : "Pendiente"; document.getElementById('q2-comision').innerText = "Q " + comision.toFixed(2); document.getElementById('q2-total-neto').innerText = "Q " + totalNeto.toFixed(2); abrirModal('modal-comisiones'); }
